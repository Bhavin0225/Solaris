# swf.fetch.cache  (stdlib + httpx + tenacity only; NO pandas import)
RETRY_STATUS = frozenset({429, 500, 502, 503, 504})

def safe_part(name: str) -> str                                    # returns name if re.fullmatch(r"[a-z0-9_]+", name) else ValueError
def is_cached(dest: Path) -> bool                                  # dest AND dest.with_suffix(".meta.json") both exist
def fetch_to_cache(client: httpx.Client, url: str, params: dict, dest: Path, *,
                   summarize: Callable[[bytes], dict], force: bool = False,
                   min_interval: float = 1.0, sleep: Callable[[float], None] = time.sleep) -> str   # "hit" | "fetched"

# swf.fetch.nasa
NASA_URL = "https://power.larc.nasa.gov/api/temporal/hourly/point"
def build_params_range(s: Settings, start: date, end: date, parameters: list[str] | None = None) -> dict   # parameters default s.data.nasa.parameters; start/end -> YYYYMMDD
def build_params(s: Settings, year: int) -> dict   # = build_params_range(s, date(year,1,1), date(year,12,31)); plan 05 probe reuses build_params_range for the NASA tail

def fetch_years(client, s: Settings, years: Iterable[int], *, root: Path, force: bool = False, sleep=time.sleep) -> list[str]
